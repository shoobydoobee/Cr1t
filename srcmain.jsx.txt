const Post = ({ post, isActive, onLike, currentUser }) => {
  const videoRef = useRef(null);
  const [isLiked, setIsLiked] = useState(false);

  useEffect(() => {
    setIsLiked(currentUser && post.likes?.includes(currentUser.uid));
  }, [post.likes, currentUser]);

  useEffect(() => {
    if (isActive && videoRef.current) {
      videoRef.current.currentTime = 0;
      videoRef.current.play().catch(e => console.log("Autoplay blocked", e));
    } else if (videoRef.current) {
      videoRef.current.pause();
    }
  }, [isActive]);

  const handleLike = () => {
    setIsLiked(!isLiked);
    onLike(post.id, !isLiked);
  };

  return (
    <div className="relative w-full h-full bg-gray-900 snap-start shrink-0 overflow-hidden">
      {post.type === 'video' ? (
        <video ref={videoRef} src={post.mediaUrl} className="w-full h-full object-cover" loop playsInline muted={false} />
      ) : (
        <img src={post.mediaUrl} className="w-full h-full object-cover" />
      )}
      <div className="absolute inset-0 bg-gradient-to-b from-black/20 via-transparent to-black/80 pointer-events-none" />
      
      <div className="absolute right-2 bottom-20 flex flex-col items-center gap-6 z-20 pb-4">
        <button onClick={handleLike} className="flex flex-col items-center gap-1">
          <div className="p-2 rounded-full bg-black/20 backdrop-blur-sm active:scale-90 transition-transform">
            <Heart size={32} className={isLiked ? "fill-red-500 text-red-500" : "text-white"} />
          </div>
          <span className="text-white text-xs font-bold shadow-black drop-shadow-md">{post.likes?.length || 0}</span>
        </button>
        <button className="flex flex-col items-center gap-1">
          <div className="p-2 rounded-full bg-black/20 backdrop-blur-sm">
            <MessageCircle size={32} className="text-white" />
          </div>
          <span className="text-white text-xs font-bold shadow-black drop-shadow-md">{post.commentCount || 0}</span>
        </button>
        <button className="p-2 rounded-full bg-black/20 backdrop-blur-sm"><Share2 size={32} className="text-white" /></button>
        <div className="w-10 h-10 rounded-full bg-gray-800 border-2 border-white overflow-hidden animate-spin-slow"><Music2 size={20} className="m-auto mt-2 text-white" /></div>
      </div>

      <div className="absolute bottom-4 left-4 right-16 z-20 text-white">
        <div className="flex items-center gap-2 mb-2">
          <div className="w-8 h-8 rounded-full bg-gradient-to-br from-yellow-400 to-orange-500 flex items-center justify-center text-xs font-bold border border-white/50">{post.userLevel || 1}</div>
          <h3 className="font-bold text-lg drop-shadow-md">@{post.username}</h3>
        </div>
        <p className="text-sm line-clamp-2 drop-shadow-md mb-2">{post.caption}</p>
        <div className="flex items-center gap-2 opacity-80"><Music2 size={12} /><p className="text-xs">Original Sound - @{post.username}</p></div>
      </div>
    </div>
  );
};

const CreatePost = ({ onPost, onCancel }) => {
  const [caption, setCaption] = useState('');
  const [mediaType, setMediaType] = useState('image');
  const [loading, setLoading] = useState(false);

  const handleSubmit = (e) => {
    e.preventDefault();
    setLoading(true);
    setTimeout(() => {
        onPost({ 
            caption, 
            type: mediaType,
            mediaUrl: mediaType === 'video' 
                ? 'https://assets.mixkit.co/videos/preview/mixkit-gaming-world-concept-1-31682-large.mp4' 
                : 'https://images.unsplash.com/photo-1542751371-adc38448a05e?auto=format&fit=crop&w=800&q=80' 
        });
        setLoading(false);
    }, 1500);
  };

  return (
    <div className="flex flex-col h-full bg-black text-white p-4">
      <div className="flex justify-between items-center mb-6">
        <button onClick={onCancel} className="text-gray-400">Cancel</button>
        <h2 className="font-bold text-lg">New Drop</h2>
        <div className="w-12"></div>
      </div>
      <div className="flex-1 flex flex-col items-center justify-center bg-gray-900 rounded-xl mb-6 relative border border-gray-800 border-dashed">
         <Camera size={48} className="text-gray-600 mb-2" />
         <p className="text-gray-500 text-sm">Tap to record or upload</p>
         <div className="absolute bottom-4 flex gap-4">
            <button onClick={() => setMediaType('video')} className={`px-4 py-1 rounded-full text-xs font-bold ${mediaType === 'video' ? 'bg-white text-black' : 'bg-black/50 text-white'}`}>VIDEO</button>
            <button onClick={() => setMediaType('image')} className={`px-4 py-1 rounded-full text-xs font-bold ${mediaType === 'image' ? 'bg-white text-black' : 'bg-black/50 text-white'}`}>PHOTO</button>
         </div>
      </div>
      <textarea value={caption} onChange={(e) => setCaption(e.target.value)} placeholder="Add a caption... #grind #gaming" className="w-full bg-gray-900 text-white p-4 rounded-xl mb-4 focus:outline-none focus:ring-2 focus:ring-blue-500 resize-none h-32" />
      <button onClick={handleSubmit} disabled={loading || !caption} className="w-full bg-blue-600 py-3 rounded-xl font-bold flex justify-center items-center gap-2">
        {loading ? <Loader2 className="animate-spin" /> : <><Send size={18} /> Post Drop</>}
      </button>
    </div>
  );
};

const Profile = ({ user, userPosts, xp, onLogout }) => {

  const nextLevel = LEVELS.find(l => l.level === currentLevel.level + 1) || { xp: 99999, title: "Max" };
  const progress = ((xp - currentLevel.xp) / (nextLevel.xp - currentLevel.xp)) * 100;

  return (
    <div className="bg-black min-h-full text-white pb-20 overflow-y-auto">
      <div className="h-32 bg-gradient-to-r from-purple-800 to-blue-600 relative">
        <div className="absolute -bottom-10 left-6">
            <div className="w-24 h-24 rounded-full border-4 border-black bg-gray-800 flex items-center justify-center text-3xl font-bold overflow-hidden">
                <img src={`https://api.dicebear.com/7.x/avataaars/svg?seed=${user?.uid}`} alt="avatar" />
            </div>
        </div>
        <button onClick={onLogout} className="absolute top-4 right-4 p-2 bg-black/30 rounded-full hover:bg-red-500/20 transition-colors"><LogOut size={20} /></button>
      </div>
      <div className="mt-12 px-6">
        <h2 className="text-2xl font-bold">@{user?.displayName || 'Gamer'}</h2>
        <div className="flex items-center gap-2 text-gray-400 text-sm mb-4">
            <span>Level {currentLevel.level} {currentLevel.title}</span>
            <span>•</span>
            <span>Joined 2024</span>
        </div>
        <div className="bg-gray-800 h-4 rounded-full overflow-hidden mb-1 relative">
            <div className="bg-gradient-to-r from-yellow-400 to-orange-500 h-full transition-all duration-1000" style={{ width: `${Math.min(progress, 100)}%` }}></div>
        </div>
        <div className="flex justify-between text-xs text-gray-500 mb-6"><span>{xp} XP</span><span>{nextLevel.xp} XP (Next Rank)</span></div>
        <div className="grid grid-cols-3 gap-4 mb-8 text-center">
            <div className="bg-gray-900 p-3 rounded-xl"><div className="font-bold text-xl">{userPosts.length}</div><div className="text-xs text-gray-500 uppercase">Drops</div></div>
            <div className="bg-gray-900 p-3 rounded-xl"><div className="font-bold text-xl">12.5k</div><div className="text-xs text-gray-500 uppercase">Followers</div></div>
            <div className="bg-gray-900 p-3 rounded-xl"><div className="font-bold text-xl">840</div><div className="text-xs text-gray-500 uppercase">Following</div></div>

                     {post.type === 'video' ? <video src={post.mediaUrl} className="w-full h-full object-cover" /> : <img src={post.mediaUrl} className="w-full h-full object-cover" />}
                </div>
            ))}
        </div>
      </div>
    </div>

};

// --- Main App Logic (Local Storage Engine) ---

const App = () => {
  const [user, setUser] = useState(null);
  const [authLoading, setAuthLoading] = useState(true);
  const [activeTab, setActiveTab] = useState('home');
  const [posts, setPosts] = useState([]);
  const [activePostIndex, setActivePostIndex] = useState(0);
  const feedContainerRef = useRef(null);
  const [xp, setXp] = useState(0);

  // 1. Initialize from Local Storage on Load
  useEffect(() => {
    // Check for user
    const savedUser = localStorage.getItem('cr1t_user');
    if (savedUser) {
      setUser(JSON.parse(savedUser));
      const savedXp = localStorage.getItem('cr1t_xp');
      setXp(savedXp ? parseInt(savedXp) : 0);
    }

    // Check for posts, or seed defaults
    const savedPosts = localStorage.getItem('cr1t_posts');
    if (savedPosts) {
      setPosts(JSON.parse(savedPosts));
    } else {
      // Seed default posts so app isn't empty
      setPosts(DEFAULT_POSTS);
      localStorage.setItem('cr1t_posts', JSON.stringify(DEFAULT_POSTS));
    }

    setAuthLoading(false);
  }, []);

  // 2. Feed Scroll Logic
  const handleScroll = () => {
    if (feedContainerRef.current) {
      const scrollPosition = feedContainerRef.current.scrollTop;
      const height = feedContainerRef.current.clientHeight;
      setActivePostIndex(Math.round(scrollPosition / height));
    }
  };

  // 3. Actions
  const handleLogin = () => {
    setAuthLoading(true);
    // Simulate API delay
    setTimeout(() => {
      const newUser = {
        uid: 'user_' + Date.now(),
        displayName: 'PlayerOne',
        isAnonymous: true
      };
      setUser(newUser);
      localStorage.setItem('cr1t_user', JSON.stringify(newUser));
      localStorage.setItem('cr1t_xp', '0');
      setXp(0);
      setAuthLoading(false);
    }, 1000);
  };

  const handleLogout = () => {
    localStorage.removeItem('cr1t_user');
    setUser(null);
    // Note: We intentionally don't clear posts so the app feels alive next time
  };

  const addXP = (amount) => {
    const newXp = xp + amount;
    setXp(newXp);
    localStorage.setItem('cr1t_xp', newXp.toString());
  };

  const handleCreatePost = (postData) => {
    const newPost = {
      ...postData,
      id: 'post_' + Date.now(),
      userId: user.uid,
      username: user.displayName,
      likes: [],
      commentCount: 0,
      userLevel: getLevel(xp).level,
      crib: "General",
      createdAt: Date.now()
    };

    const updatedPosts = [newPost, ...posts];
    setPosts(updatedPosts);
    localStorage.setItem('cr1t_posts', JSON.stringify(updatedPosts));
    
    addXP(50);
    setActiveTab('home');
  };

  const handleLike = (postId, isLiked) => {
    if (!user) return;
    
    const updatedPosts = posts.map(post => {
      if (post.id === postId) {
        let newLikes = post.likes || [];
        if (isLiked) {
          if (!newLikes.includes(user.uid)) newLikes.push(user.uid);
        } else {
          newLikes = newLikes.filter(uid => uid !== user.uid);
        }
        return { ...post, likes: newLikes };
      }
      return post;
    });

    setPosts(updatedPosts);
    localStorage.setItem('cr1t_posts', JSON.stringify(updatedPosts));

    if (isLiked) addXP(10);
  };

  if (authLoading) return <div className="h-screen bg-black flex items-center justify-center text-white"><Loader2 className="animate-spin" /></div>;

  if (!user) return <AuthScreen onLogin={handleLogin} isLoading={authLoading} />;

  return (
    <div className="h-screen w-full bg-black flex flex-col overflow-hidden font-sans select-none">
      <div className="flex-1 relative overflow-hidden">
        {activeTab === 'home' && (
            <div ref={feedContainerRef} onScroll={handleScroll} className="h-full w-full overflow-y-scroll snap-y snap-mandatory scrollbar-hide">
                {posts.length === 0 ? (
                    <div className="h-full w-full flex flex-col items-center justify-center text-gray-500 gap-4">
                        <Flame size={48} />
                        <p>Feed is empty. Be the first to drop!</p>
                        <button onClick={() => setActiveTab('create')} className="bg-blue-600 text-white px-6 py-2 rounded-full font-bold">Create Post</button>
                    </div>
                ) : (
                    posts.map((post, index) => (
                        <div key={post.id} className="h-full w-full snap-start">
                            <Post post={post} isActive={index === activePostIndex} onLike={handleLike} currentUser={user} />
                        </div>
                    ))
                )}
            </div>
        )}
        {activeTab === 'create' && <div className="absolute inset-0 z-40 bg-black"><CreatePost onPost={handleCreatePost} onCancel={() => setActiveTab('home')} /></div>}
        {activeTab === 'profile' && <div className="h-full overflow-y-auto bg-black"><Profile user={user} userPosts={posts.filter(p => p.userId === user.uid)} xp={xp} onLogout={handleLogout} /></div>}
        {activeTab === 'cribs' && <div className="h-full overflow-y-auto bg-black p-4 text-white"><h1 className="text-2xl font-bold mb-4">Cribs</h1><p>Community features coming soon.</p></div>}
        
        <div className="absolute top-4 right-4 z-30 bg-black/50 backdrop-blur-md border border-yellow-500/30 px-3 py-1 rounded-full flex items-center gap-2 text-yellow-400 font-bold text-sm pointer-events-none">
            <Zap size={14} className="fill-yellow-400" /><span>{xp} XP</span>
        </div>
      </div>
      <NavBar activeTab={activeTab} onTabChange={setActiveTab} />
    </div>
  );
};

// --- Bootstrapping ---
const rootElement = document.getElementById('root');
if (rootElement) {
  rootElement.innerHTML = '';
  const root = createRoot(rootElement);
  root.render(<App />);
}
